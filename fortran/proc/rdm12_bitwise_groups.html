rdm12_bitwise_groups

public subroutine rdm12_bitwise_groups(ndet, norb, nw, wa, wb, c, nint, ng, gstart, grp, nnb, nbstart, nb, d1, d2, nthreads, ierr)

The same pair loop restricted to candidate pairs: determinants are sorted into groups (contiguous, gstart(g):gstart(g+1)-1), and for group g only the groups nb(nbstart(g):nbstart(g+1)-1), all >= g, are paired with it (g itself first when it is to be included). Elements of dm2 with all four orbitals >= nint are not accumulated (they are formed elsewhere); nint < 0 keeps them. The (I, J) / (J, I) symmetrisation is applied as in rdm12_bitwise, so pairs must be listed once (g <= g').

    subroutine rdm12_bitwise_groups(ndet, norb, nw, wa, wb, c, nint, ng, gstart, grp, nnb, nbstart, nb, &
                                    d1, d2, nthreads, ierr)
        integer, intent(in) :: ndet, norb, nw, nint, ng, nnb, nthreads
        integer(i8), intent(in) :: wa(nw, ndet), wb(nw, ndet)
        real(dp), intent(in) :: c(ndet)
        integer, intent(in) :: gstart(ng + 1), grp(ndet), nbstart(ng + 1), nb(nnb)
        real(dp), intent(out), target :: d1(norb, norb), d2(norb, norb, norb, norb)
        integer, intent(out) :: ierr
        real(dp), allocatable, target :: a1(:, :), a2(:, :, :, :)
        type(accum) :: acc
        integer :: i, j, nd, nthr, p, q, r, s, iw, g, k, g2, j0, j1
        real(dp) :: bytes, tmp
        logical :: atomic

        ierr = 0
        if (norb < 1 .or. nw < (norb + 63) / 64 .or. ng < 1) then
            ierr = 1
            return
        end if
        d1 = 0.0_dp
        d2 = 0.0_dp
        bytes = 8.0_dp * real(norb, dp)**4
        atomic = bytes * real(max(1, nthreads), dp) > 2.0e9_dp
        nthr = max(1, nthreads)
        !$omp parallel num_threads(nthr) default(shared) private(a1, a2, acc, i, j, nd, iw, g, k, g2, j0, j1)
        acc%atomic = atomic
        acc%nint = nint
        if (atomic) then
            acc%a1 => d1
            acc%a2 => d2
        else
            allocate(a1(norb, norb), a2(norb, norb, norb, norb))
            a1 = 0.0_dp
            a2 = 0.0_dp
            acc%a1 => a1
            acc%a2 => a2
        end if
        !$omp do schedule(dynamic, 16)
        do i = 1, ndet
            call diagonal(wa(:, i), wb(:, i), nw, 0.5_dp * c(i) * c(i), acc)
            g = grp(i)
            do k = nbstart(g), nbstart(g + 1) - 1
                g2 = nb(k)
                j0 = gstart(g2)
                if (g2 == g) j0 = i + 1
                j1 = gstart(g2 + 1) - 1
                do j = j0, j1
                    nd = 0
                    do iw = 1, nw
                        nd = nd + popcnt(ieor(wa(iw, i), wa(iw, j))) + popcnt(ieor(wb(iw, i), wb(iw, j)))
                    end do
                    if (nd > 4) cycle
                    if (nd == 2) then
                        call single(wa(:, i), wb(:, i), wa(:, j), wb(:, j), nw, c(i) * c(j), acc)
                    else if (nd == 4) then
                        call double(wa(:, i), wb(:, i), wa(:, j), wb(:, j), nw, c(i) * c(j), acc)
                    end if
                end do
            end do
        end do
        !$omp end do
        if (.not. atomic) then
            !$omp critical
            d1 = d1 + a1
            d2 = d2 + a2
            !$omp end critical
            deallocate(a1, a2)
        end if
        !$omp end parallel
        d1 = d1 + transpose(d1)
        !$omp parallel do default(shared) private(p, q, r, s, tmp) schedule(dynamic)
        do p = 1, norb
            do q = 1, norb
                do r = 1, norb
                    do s = 1, norb
                        if (lower2(p, q, r, s)) then
                            tmp = d2(s, r, q, p) + d2(r, s, p, q)
                            d2(s, r, q, p) = tmp
                            d2(r, s, p, q) = tmp
                        else if (p == q .and. r == s) then
                            d2(s, r, q, p) = 2.0_dp * d2(s, r, q, p)
                        end if
                    end do
                end do
            end do
        end do
        !$omp end parallel do
    contains
        !> (p,q,r,s) strictly before its partner (q,p,s,r) in lexicographic order
        pure logical function lower2(p, q, r, s)
            integer, intent(in) :: p, q, r, s
            if (p /= q) then
                lower2 = p < q
            else
                lower2 = r < s
            end if
        end function lower2
    end subroutine rdm12_bitwise_groups
