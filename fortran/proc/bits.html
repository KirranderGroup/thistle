    subroutine bits(wa, wb, nw, orb, spn, n)
        integer, intent(in) :: nw
        integer(i8), intent(in) :: wa(nw), wb(nw)
        integer, intent(out) :: orb(:), spn(:), n
        integer(i8) :: w
        integer :: iw
        n = 0
        do iw = 1, nw
            w = wa(iw)
            do while (w /= 0_i8)
                n = n + 1
                orb(n) = 64 * (iw - 1) + trailz(w)
                spn(n) = 0
                w = iand(w, w - 1_i8)
            end do
        end do
        do iw = 1, nw
            w = wb(iw)
            do while (w /= 0_i8)
                n = n + 1
                orb(n) = 64 * (iw - 1) + trailz(w)
                spn(n) = 1
                w = iand(w, w - 1_i8)
            end do
        end do
    end subroutine bits
